Código utilizado (de dónde salen estas cifras) 
import pandas as pd

df = pd.read_csv("data/raw/vehiculos_electricos_hibridos_runt.csv",
                 dtype=str, keep_default_na=False, na_values=[""])
df["cantidad"] = df["cantidad"].astype(int)

total = df["cantidad"].sum()                                   # {{ p.vehiculos | miles }} vehículos
por_combustible = df.groupby("combustible")["cantidad"].sum().sort_values(ascending=False)
participacion = (por_combustible / total * 100).round(2)       # % por tecnología
por_clase = df.groupby("clase")["cantidad"].sum().sort_values(ascending=False)
por_carroceria = df.groupby("carroceria")["cantidad"].sum().sort_values(ascending=False)
