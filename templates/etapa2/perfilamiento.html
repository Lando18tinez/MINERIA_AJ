df = pd.read_csv("data/raw/vehiculos_electricos_hibridos_runt.csv",
                 dtype=str, keep_default_na=False, na_values=[""])
df["cantidad"] = df["cantidad"].astype(int)

len(df), df["cantidad"].sum()             # {{ res.meta.filas | miles }} filas, {{ res.meta.vehiculos | miles }} vehículos
